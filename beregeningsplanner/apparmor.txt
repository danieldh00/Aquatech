#include <tunables/global>

# AppArmor-profiel voor de Beregeningsplanner-add-on. De container draait een
# kaal Node-proces (geen s6-overlay, geen bashio) dat alleen statische
# bestanden serveert; opslag gebeurt (nog) in de browser.
profile beregeningsplanner flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  file,
  signal,

  /usr/local/bin/node rix,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/local/bin/** ix,
  /run/** rwk,
  /tmp/** rw,
  /data/** rw,

  capability chown,
  capability dac_override,
  capability fowner,
  capability kill,
  capability setgid,
  capability setuid,

  deny mount,
  deny umount,
  deny ptrace,
  deny @{PROC}/sys/kernel/** w,
}
